** word completion
Dictionary completion in text modes via look(1) over =dict/words.txt=
(rebuild with =dict/build.sh=).
#+begin_src emacs-lisp
(defvar +dict-file (expand-file-name "dict/words.txt" doom-user-dir)
  "Lowercased, LC_ALL=C sorted word list used by `+dict-capf'.")

(defvar +dict-limit 200
  "Maximum number of candidates returned by `+dict-capf'.")

(defun +dict-capf ()
  "Complete word at point from `+dict-file', keeping the typed prefix's case."
  (when-let* ((bounds (bounds-of-thing-at-point 'word))
              ((< (car bounds) (point)))
              ((file-readable-p +dict-file)))
    (let* ((beg (car bounds))
           (prefix (buffer-substring-no-properties beg (point)))
           (len (length prefix)))
      (list beg (cdr bounds)
            (mapcar (lambda (word) (concat prefix (substring word len)))
                    (take +dict-limit
                          (process-lines-ignore-status
                           "look" (downcase prefix) +dict-file)))
            :exclusive 'no))))

;; Replaces the built-in `ispell-completion-at-point'
(setq text-mode-ispell-word-completion nil)

(add-hook! 'text-mode-hook
  (defun +dict-add-capf-h ()
    (add-hook 'completion-at-point-functions
              (cape-capf-buster #'+dict-capf) 30 t)))
#+end_src

** Fix latex previewing:
#+begin_src emacs-lisp
;; org-latex-compiler is xelatex, so the preview header pulls in fontspec,
;; which the stock dvisvgm process (plain latex) can't load. Compile previews
;; with xelatex to .xdv instead and convert that with dvisvgm.
(after! org
  (add-to-list 'org-preview-latex-process-alist
               '(xdvisvgm
                 :programs ("xelatex" "dvisvgm")
                 :description "xdv > svg"
                 :message "you need to install xelatex and dvisvgm."
                 :image-input-type "xdv"
                 :image-output-type "svg"
                 :image-size-adjust (1.7 . 1.5)
                 :latex-compiler ("xelatex -no-pdf -interaction nonstopmode -output-directory %o %f")
                 :image-converter ("dvisvgm %f --no-fonts --exact-bbox --scale=%S --output=%O")))
  (setq org-preview-latex-default-process 'xdvisvgm)
  ;; Transparent: the default bg is taken from the global default face, which
  ;; doesn't match the buffer's background and leaves a visible box.
  (setq org-format-latex-options
        (plist-put (plist-put org-format-latex-options :scale 2)
                   :background "Transparent")
        ;; The pretty module turns this on; toggle with SPC t p instead.
        org-pretty-entities nil)

  ;; Native LaTeX highlighting paints environments with the extended org-block
  ;; background, which shows through the transparent preview as a full-width
  ;; bar. Mask it with the default face, only while the preview is shown.
  (defadvice! my/org-latex-preview-mask-block-bg-a (beg end &rest _)
    :after #'org--make-preview-overlay
    (dolist (ov (overlays-at beg))
      (when (eq (overlay-get ov 'org-overlay-type) 'org-latex-overlay)
        (overlay-put ov 'face 'default)))
    ;; The bar past the image comes from the trailing newline's face.
    (when (eq (char-after end) ?\n)
      (let ((ov (make-overlay end (1+ end))))
        (overlay-put ov 'org-overlay-type 'org-latex-overlay)
        (overlay-put ov 'evaporate t)
        (overlay-put ov 'face '(:inherit default :extend t))))))
#+end_src

(defun my/gost-titlepage-latex (work discipline topic students checker-pos checker-name city year)
  "Build the GOST titlepage LaTeX export block.
STUDENTS is a list of (GROUP . NAME) conses."
  (concat
   "#+begin_export latex\n"
   "\\begin{titlepage}\n"
   "\\begin{center}\n"
   "МИНОБРНАУКИ РОССИИ\\\\\n"
   "Федеральное государственное бюджетное образовательное учреждение\\\\\n"
   "высшего образования\\\\\n"
   "«Ижевский государственный технический университет имени М.\\,Т.\\ Калашникова»\\\\\n"
   "Факультет «Информационные технологии, радиоэлектроника и приборостроение»\\\\\n"
   "Кафедра «Программное обеспечение»\n"
   "\\end{center}\n\n"
   "\\vspace{4cm}\n\n"
   "\\begin{center}\n"
   "Отчет\\\\\n"
   (or work "")
   (if (or (null discipline) (string-empty-p discipline)) "" (concat "\\\\\nпо дисциплине «" discipline "»"))
   (if (or (null topic) (string-empty-p topic)) "" (concat "\\\\\nна тему «" topic "»"))
   "\n\\end{center}\n\n"
   "\\vfill\n\n"
   "\\noindent " (if (= (length students) 1) "Выполнил" "Выполнили") ":\\\\[2mm]\n"
   (let ((n (length students))
         (i 0))
     (mapconcat (lambda (s)
                  (setq i (1+ i))
                  (concat "студент гр. " (car s) " \\hfill " (cdr s)
                          (if (= i n) "\\\\[4mm]" "\\\\")))
                students "\n"))
   "\nПроверил:\\\\[2mm]\n"
   (or checker-pos "") " \\hfill " (or checker-name "") "\\\\\n\n"
   "\\vfill\n\n"
   "\\begin{center}\n"
   (or city "") " " (or year "") "\n"
   "\\end{center}\n"
   "\\end{titlepage}\n"
   "#+end_export\n"))

;;; #+KEYWORD lines filled in by hand, expanded on export
(defun my/gost-kw (name)
  "Value of the first #+NAME: keyword line in the current buffer."
  (save-excursion
    (goto-char (point-min))
    (when (re-search-forward (format "^#\\+%s:[ \t]*\\(.*\\)$" (regexp-quote name)) nil t)
      (match-string-no-properties 1))))

(defun my/gost-kw-all (name)
  "Values of every #+NAME: keyword line in the current buffer."
  (save-excursion
    (goto-char (point-min))
    (let (res)
      (while (re-search-forward (format "^#\\+%s:[ \t]*\\(.*\\)$" (regexp-quote name)) nil t)
        (push (match-string-no-properties 1) res))
      (nreverse res))))

(defun my/gost-split-pair (s)
  "Split \"a | b\" into (\"a\" . \"b\"), trimmed."
  (let ((parts (split-string (or s "") "|")))
    (cons (string-trim (or (nth 0 parts) "")) (string-trim (or (nth 1 parts) "")))))

(defun my/gost-export-titlepage (backend)
  "Replace a #+GOST_TITLEPAGE: line with a titlepage built from #+TYPE/#+DISCIPLINE/... keywords."
  (when (org-export-derived-backend-p backend 'latex)
    (goto-char (point-min))
    (when (re-search-forward "^#\\+GOST_TITLEPAGE:.*$" nil t)
      (let ((beg (match-beginning 0))
            (end (match-end 0))
            (checker (my/gost-split-pair (my/gost-kw "CHECKER"))))
        (let ((replacement
               (my/gost-titlepage-latex
                (my/gost-kw "TYPE")
                (my/gost-kw "DISCIPLINE")
                (my/gost-kw "TOPIC")
                (mapcar #'my/gost-split-pair (my/gost-kw-all "STUDENT"))
                (car checker) (cdr checker)
                (my/gost-kw "CITY")
                (my/gost-kw "YEAR"))))
          (delete-region beg end)
          (goto-char beg)
          (insert replacement))))))

(add-hook 'org-export-before-parsing-hook #'my/gost-export-titlepage)

** Stale session envvars
~doom sync --env~ snapshots the shell environment into ~05-doom-env.load.el~ and
prepends it to ~process-environment~. Session-specific variables in that snapshot
go stale after the next login and shadow the values the daemon actually
inherited, so ~hyprctl~ talks to a dead socket from eshell, vterm and ghostel.
Drop the duplicates and let the inherited values through.
#+begin_src emacs-lisp
(defun +my/unshadow-envvar (var)
  "Drop stale duplicate entries of VAR from `process-environment'.
The envvar snapshot is prepended to the environment Emacs inherited, so a
variable can appear more than once; keep only the last (inherited) entry."
  (let* ((re (concat "\\`" (regexp-quote var) "\\(=\\|\\'\\)"))
         (env (default-value 'process-environment))
         (matches (seq-filter (lambda (e) (string-match-p re e)) env))
         (keep (car (last matches))))
    (when (cdr matches)
      (setq-default process-environment
                    (seq-remove (lambda (e)
                                  (and (string-match-p re e)
                                       (not (eq e keep))))
                                env)))))

(mapc #'+my/unshadow-envvar
      '("HYPRLAND_INSTANCE_SIGNATURE" "HYPRLAND_CMD" "HL_INITIAL_WORKSPACE_TOKEN"
        "WAYLAND_DISPLAY" "XDG_ACTIVATION_TOKEN" "TMUX" "TMUX_PANE" "COLUMNS"))
#+end_src
** Double space inserts a period
Like on macOS and phones: two quick spaces after a word turn into ". ".
Only in text buffers, and not inside org src blocks.
#+begin_src emacs-lisp
(defvar +my/double-space-period-delay 0.5
  "Max seconds between two spaces for them to become a period.")

(defvar-local +my/double-space-period--last nil
  "Time of the last space typed, or nil.")

(defun +my/double-space-period--maybe ()
  "Replace \"word  \" with \"word. \" when two spaces are typed quickly."
  (if (not (eq last-command-event ?\s))
      (setq +my/double-space-period--last nil)
    (let ((last +my/double-space-period--last))
      (if (and last
               (< (float-time (time-since last)) +my/double-space-period-delay)
               (looking-back "[[:alnum:])»\"'”]  " (- (point) 3))
               (not (and (derived-mode-p 'org-mode) (org-in-src-block-p))))
          (progn
            (delete-char -2)
            (insert ". ")
            (setq +my/double-space-period--last nil))
        (setq +my/double-space-period--last (current-time))))))

(define-minor-mode +my/double-space-period-mode
  "Insert a period when space is typed twice quickly."
  :lighter nil
  (if +my/double-space-period-mode
      (add-hook 'post-self-insert-hook #'+my/double-space-period--maybe nil t)
    (remove-hook 'post-self-insert-hook #'+my/double-space-period--maybe t)))

(add-hook 'text-mode-hook #'+my/double-space-period-mode)
#+end_src
